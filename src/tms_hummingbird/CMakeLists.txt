cmake_minimum_required(VERSION 3.8)
project(tms_hummingbird)

if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
if(CMAKE_COMPILER_IS_GNUCXX OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(px4_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(dynamixel_sdk REQUIRED)

add_executable(bldc_control src/bldc_control.cpp)
add_executable(dynamixel_control src/dynamixel_control.cpp)
target_include_directories(bldc_control PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_include_directories(dynamixel_control PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(dynamixel_control dynamixel_sdk::dynamixel_sdk)
ament_target_dependencies(bldc_control rclcpp px4_msgs std_msgs geometry_msgs)
ament_target_dependencies(dynamixel_control rclcpp std_msgs)
install(TARGETS bldc_control dynamixel_control DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/tms_gui.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch include DESTINATION share/${PROJECT_NAME})
ament_package()
